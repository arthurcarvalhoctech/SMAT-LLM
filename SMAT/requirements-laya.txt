# Optional dependency for the Laya pre-analysis flow.
#
# Python note: use a final Python 3.11.x, not Ubuntu 22.04's python3.11
# (3.11.0rc1). That interpreter lacks sys.get_int_max_str_digits and breaks
# torch._dynamo with a misleading "Duplicate dispatch rule" error. The
# project Dockerfile installs Python 3.11 via deadsnakes and guards the build.
#
# transformers 4.52+ pulls integrations that interact badly with some
# torch._dynamo versions. 4.51.3 is the version validated with laya 0.4.12
# (a live CPU smoke test completed in ~9 s with max_len=1024).
transformers>=4.48,<4.52
laya>=0.4.12,<0.5

# Torch is installed separately by the Dockerfile from TORCH_INDEX_URL
# (CPU by default), so `laya` does not pull a CUDA wheel. For a local install,
# install the CPU wheel first:
#   python -m pip install torch --index-url https://download.pytorch.org/whl/cpu
